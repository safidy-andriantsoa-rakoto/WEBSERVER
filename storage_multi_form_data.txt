  <script>
    const $ = (s) => document.querySelector(s);
    const reqOut = $('#reqOut');
    const resOut = $('#resOut');
    const statusTag = $('#statusTag');
    const timeTag = $('#timeTag');
    const sizeTag = $('#sizeTag');

    function setMeta(status, elapsed, size) {
      statusTag.textContent = 'Status: ' + (status ?? '—');
      timeTag.textContent = 'Durée: ' + (elapsed != null ? elapsed + ' ms' : '—');
      sizeTag.textContent = 'Taille: ' + (size != null ? size + ' o' : '—');
    }

    function showReq(method, url, headers, body) {
      const h = Object.entries(headers || {}).map(([k,v]) => k+': '+v).join('\\n');
      reqOut.textContent = [method + ' ' + url, h, '', body || ''].join('\\n');
    }

    async function doDelete(url, { jsonBody, timeoutMs }) {
      const controller = new AbortController();
      let tId;
      if (timeoutMs > 0) tId = setTimeout(() => controller.abort(), timeoutMs);

      const headers = {};
      let body;
      if (jsonBody) {
        headers['Content-Type'] = 'application/json';
        body = JSON.stringify(jsonBody);
      }

      showReq('DELETE', url, headers, body);

      const started = performance.now();
      let status, text = '', size;
      try {
        const res = await fetch(url, { method: 'DELETE', headers, body, signal: controller.signal, mode: 'cors' });
        status = res.status + ' ' + res.statusText;
        const ab = await res.arrayBuffer();
        size = ab.byteLength;
        text = new TextDecoder().decode(ab);
      } catch (e) {
        text = 'Erreur: ' + (e?.message || e);
      } finally {
        clearTimeout(tId);
      }
      const elapsed = Math.round(performance.now() - started);
      setMeta(status, elapsed, size);
      resOut.textContent = text;
    }

    $('#btnDelete').addEventListener('click', async () => {
      const base = $('#apiUrl').value.trim();
      const path = $('#filePath').value.trim();
      const mode = $('#mode').value;
      const timeout = Number($('#timeout').value || 0);

      if (!base) { alert('Indiquez l’URL de l’API.'); return; }
      if (!path) { alert('Indiquez le chemin du fichier.'); return; }

      let url = base;
      let jsonBody = null;

      if (mode === 'param') {
        // supporte /files/:name ou ?path=...
        if (base.includes(':name')) {
          url = base.replace(':name', encodeURIComponent(path));
        } else if (base.includes('?') || base.endsWith('.php')) {
          const sep = base.includes('?') ? '&' : '?';
          url = base + sep + 'path=' + encodeURIComponent(path);
        } else if (base.endsWith('/files')) {
          url = base + '/' + encodeURIComponent(path);
        } else {
          // par défaut, on ajoute ?path=
          const sep = base.includes('?') ? '&' : '?';
          url = base + sep + 'path=' + encodeURIComponent(path);
        }
        await doDelete(url, { jsonBody: null, timeoutMs: timeout });
      } else {
        // envoi dans le corps JSON
        url = base;
        jsonBody = { path };
        await doDelete(url, { jsonBody, timeoutMs: timeout });
      }
    });
